# ---- --help-Texte (help.<Optionsname>)
help.about = `ls`, mit Bildern.\nEin Auflistungswerkzeug im Stil von `ls`, das im Terminal Vorschaubilder von Bildern, Videos und PDFs zeigt, mit Sixel-/Kitty-/iTerm2-Grafik, Halbblock-Zeichen oder ASCII-Art.\nMehrere Dateien, Verzeichnisse oder Platzhalter ergeben eine Vorschauliste mit Dateinamen.
help.images = Bilddateien, Verzeichnisse oder Platzhalter (z. B. ./*). Standard: das aktuelle Verzeichnis
help.regexp = Nach Dateiname (ohne Verzeichnisteil) mit einem regulären Ausdruck filtern. Mehrfach angeben, damit eines von mehreren Mustern passt. Beispiel: -e "\.png$" -e "(?i)^img_\d+" ((?i) ignoriert Groß-/Kleinschreibung)
help.recursive = Auch Unterverzeichnisse durchsuchen (versteckte Verzeichnisse werden übersprungen). Der Name des übergeordneten Verzeichnisses steht vor dem Dateinamen
help.max_depth = Wie tief abgestiegen wird (0: nur das angegebene Verzeichnis, 1: eine Ebene tiefer). Impliziert -R
help.sort = Sortierschlüssel. name ist aufsteigend; date, size und exif-date beginnen mit dem Neuesten / Größten (-r kehrt um)
help.reverse = Reihenfolge umkehren
help.head = Nach dem Sortieren nur die ersten N Bilder anzeigen (z. B. --sort date -n 20 für die 20 neuesten)
help.width = Ausgabebreite in Spalten: die Bildbreite bei einem einzelnen Bild, die Gesamtbreite bei einer Liste. Nicht mit --size kombinierbar
help.size = Größenvorgabe (Standard: s). Einzelnes Bild: 24/40/80/120/160 Spalten; Liste: jede Vorschau 12/20/40/60/80 Spalten (xs/s/m/l/xl)
help.palette = Voreingestellter Zeichensatz für den Modus text
help.custom_palette = Beliebiger Zeichensatz, von dunkel nach hell (hat Vorrang vor --palette)
help.invert = Helligkeit invertieren
help.no_color = ANSI-Farben abschalten (einfarbig)
help.no_links = Dateinamen keine Hyperlinks (Strg+Klick zum Öffnen) hinzufügen
help.no_pager = Nicht anhalten (wie more), wenn die Ausgabe nicht auf den Bildschirm passt
help.no_cache = Den Render-Cache nicht verwenden
help.cache_max_mb = Größenlimit des Caches in MB (Standard: 256 oder die Umgebungsvariable GLS_CACHE_MAX_MB). Darüber hinausgehende alte Einträge werden sofort entfernt
help.cache_days = Cache-Einträge löschen, die so viele Tage nicht verwendet wurden (Standard: 30 oder die Umgebungsvariable GLS_CACHE_DAYS)
help.cache_info = Zeigt, wo der Cache liegt, wie groß er ist und welche Limits gelten, und beendet das Programm
help.clear_cache = Löscht den gesamten Render-Cache und beendet das Programm
help.contrast = Kontrast (1.0: unverändert; größer ist kräftiger)
help.gamma = Gamma (1.0: unverändert; über 1 heller, unter 1 dunkler)
help.mode = Darstellungsmodus. Standard: image (Bildprotokoll des Terminals), je nach Terminal mit Rückfall auf half (TrueColor, dann 256 Farben) und dann text (einfarbig). half lässt sich nicht mit --no-color kombinieren (dann wird text verwendet). -p/--palette gilt nur für text
help.protocol = Bildprotokoll für --mode image. auto erkennt es anhand von Umgebungsvariablen
help.cell_size = Pixelgröße einer Zeichenzelle (z. B. 10x20). Anpassen, wenn Bilder und Text bei Sixel nicht übereinanderliegen
help.grid = Raster einer Liste (COLSxROWS pro Seite); lange Listen werden auf Seiten verteilt. Standard: so viele Spalten, wie in die Terminalbreite passen (2 Zeilen); 3x2, wenn nur --width angegeben ist
help.verbose = Bildinfos unter jedem Namen anzeigen. -v: Pixelgröße und Dateigröße; -vv: zusätzlich Format, Farbe, Megapixel, Seitenverhältnis, Änderungsdatum und EXIF (Kamera und Aufnahmeeinstellungen)
help.long = Lange Ausgabe wie bei ls -l: je Datei Größe, Abmessungen, Format, Datum, Kamera und Aufnahmeeinstellungen (EXIF) und mehr, im Terminal mit Vorschaubild. In einer Pipe: eine schlichte Zeile pro Datei. Spalten ohne Inhalt entfallen
help.thumbs_off = Zeigt mit -l keine Vorschaubilder, sondern nur die Tabelle mit einer Zeile pro Datei (das gilt auch, wenn die Ausgabe in eine Pipe geleitet wird)
help.json = Gibt die passenden Dateien als JSON aus (ein Array mit einem Objekt pro Datei: Pfad, Größe, Abmessungen, Format, Datum, EXIF) und beendet das Programm. Für Skripte
help.null = Gibt nur die Pfade der passenden Dateien aus, getrennt durch NUL-Zeichen (für xargs -0), und beendet das Programm
help.lang = Sprache der Meldungen (z. B. en, ja). Standard: aus der Umgebung erkannt
