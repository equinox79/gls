# ---- texte de --help (help.<nom de l'option>)
help.about = `ls` pour les images : affiche des images dans le terminal avec un protocole d'image (Sixel / Kitty / iTerm2), des demi-blocs ou de l'art ASCII.\nIndiquez plusieurs images (jokers et répertoires acceptés) pour obtenir une liste de miniatures avec les noms de fichiers.
help.images = Fichiers image, répertoires ou jokers (p. ex. ./*). Par défaut : le répertoire courant
help.regexp = Filtre par nom de fichier (sans le répertoire) avec une expression régulière. À répéter pour accepter l'un de plusieurs motifs. Exemple : -e "\.png$" -e "(?i)^img_\d+" ((?i) ignore la casse)
help.recursive = Cherche aussi dans les sous-répertoires (les répertoires cachés sont ignorés). Le nom du répertoire parent est affiché avant le nom du fichier
help.max_depth = Profondeur de descente (0 : seulement le répertoire indiqué, 1 : un niveau de plus). Implique -R
help.sort = Critère de tri. name est croissant ; date, size et exif-date vont du plus récent / plus gros au plus ancien / plus petit (-r inverse)
help.reverse = Inverse l'ordre
help.head = Après le tri, n'affiche que les N premières images (p. ex. --sort date -n 20 pour les 20 plus récentes)
help.width = Largeur de sortie en colonnes : la largeur de l'image pour une seule image, la largeur totale pour une liste. Incompatible avec --size
help.size = Taille prédéfinie (par défaut : s). Une image : 24/40/80/120/160 colonnes ; liste : chaque miniature 12/20/40/60/80 colonnes (xs/s/m/l/xl)
help.palette = Jeu de caractères prédéfini pour le mode text
help.custom_palette = Jeu de caractères au choix, du plus sombre au plus clair (prioritaire sur --palette)
help.invert = Inverse la luminosité
help.no_color = Désactive les couleurs ANSI (monochrome)
help.no_links = N'ajoute pas d'hyperliens (Ctrl+clic pour ouvrir) aux noms de fichiers
help.no_pager = Ne fait pas de pause (comme more) quand la sortie dépasse l'écran
help.no_cache = N'utilise pas le cache de rendu
help.cache_max_mb = Taille maximale du cache en Mo (par défaut : 256, ou la variable d'environnement GLS_CACHE_MAX_MB). Les plus anciennes entrées au-delà sont supprimées immédiatement
help.cache_days = Supprime les entrées du cache inutilisées depuis ce nombre de jours (par défaut : 30, ou la variable d'environnement GLS_CACHE_DAYS)
help.cache_info = Affiche l'emplacement du cache, sa taille et ses limites, puis quitte
help.clear_cache = Supprime tout le cache de rendu, puis quitte
help.contrast = Contraste (1.0 : inchangé ; plus grand donne plus de contraste)
help.gamma = Gamma (1.0 : inchangé ; au-dessus de 1, plus clair ; en dessous de 1, plus sombre)
help.mode = Mode de rendu. Par défaut : image (protocole d'image du terminal), avec repli sur half (TrueColor, puis 256 couleurs) puis sur text (monochrome) selon le terminal. half est incompatible avec --no-color (text est alors utilisé). -p/--palette ne s'applique qu'à text
help.protocol = Protocole d'image pour --mode image. auto le détecte à partir des variables d'environnement
help.cell_size = Taille en pixels d'une cellule de caractère (p. ex. 10x20). À ajuster si les images et le texte sont décalés avec Sixel
help.grid = Grille d'une liste (COLSxROWS par page) ; les longues listes sont réparties en pages. Par défaut, autant de colonnes que la largeur du terminal en permet (2 lignes) ; 3x2 si seul --width est donné
help.verbose = Affiche des informations sur l'image sous chaque nom. -v : taille en pixels et taille du fichier ; -vv : aussi le format, la couleur, les mégapixels, le rapport d'aspect, la date de modification et l'EXIF (appareil et réglages de prise de vue)
help.long = Liste détaillée, comme ls -l : une ligne par fichier avec taille, dimensions, format, dates, appareil et réglages de prise de vue (EXIF) et plus. Les colonnes sans rien à afficher sont omises
help.lang = Langue des messages (p. ex. en, ja). Par défaut : détectée à partir de l'environnement
